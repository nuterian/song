# Pinned for macOS arm64 + Python 3.12, installed into visuals/.venv.
#
# numpy is held below 2.0 for the same reason the root requirements hold it
# there: milestone 2 brings in torch and librosa, and the 1.26 ABI is what that
# combination is validated against. Pinning it now means the venv does not have
# to be rebuilt then.
numpy==1.26.4

# Headless OpenGL. On macOS moderngl's standalone context comes up on CGL and
# reports "4.1 Metal", which is the newest desktop GL Apple ships; see the
# shader-dialect note in DESIGN.md for why that is enough to run GLSL ES 3.0
# shaders written for WebGL2.
moderngl==5.12.0

# Listening, for pieces (visuals/pieces/): zero-phase filters and envelopes, vocal
# pitch, constant-Q. The same librosa the root requirements pin. Stem separation
# itself is run once from the root venv, which is the only place torch lives.
scipy==1.14.1
librosa==0.10.2.post1
soundfile==0.12.1

# Small local models, for pieces (visuals/pieces/gravity/models.py). All permissive
# licences. torch is 2.6 rather than the root's 2.5.1 because transformers will not
# torch.load a pickle checkpoint on anything older (CVE-2025-32434), and CLAP's
# only checkpoint is a pickle. basic-pitch itself is not installed - its pip package
# stops at Python 3.11 - only its 230 KB nmp.onnx, fetched to visuals/cache/models/
# from github.com/spotify/basic-pitch at tag v0.4.0 and run with onnxruntime.
torch==2.6.0
torchaudio==2.6.0
onnxruntime==1.30.0
transformers==4.57.6
swift-f0==0.2.0
beat-this==1.1.0
lv-chordia==1.1.0

# The words, burned into an mp4 (pieces/gravity/burn.py): Pillow sets them in the
# player's own font file, visuals/player/fonts/Inter-Light.woff2.
Pillow==12.3.0

pytest==8.3.4
